## What is a Local Secondary Index (LSI)?

It’s an index we define that shares the table's partition key but uses an alternate sort key for scans and queries. It must be created at table creation time you cannot add, remove, or modify it later. All queries based on that sort key are significantly faster using the index than the main table.

function updatePartitionOptions() {

  /*
   * Get every unique UserId from the data.
   */

  const partitionKeys = [
    ...new Set(
      users.map(user => user.UserId)
    )
  ];


  /*
   * Remember the currently selected
   * value if possible.
   */

  const previousValue =
    queryPartition.value;


  /*
   * Clear the dropdown.
   */

  queryPartition.innerHTML = "";


  /*
   * Create an option for every
   * partition key.
   */

  partitionKeys.forEach(userId => {

    const option =
      document.createElement("option");

    option.value =
      userId;

    option.textContent =
      userId;

    queryPartition.appendChild(
      option
    );

  });


  /*
   * Keep the previous selection if
   * that UserId still exists.
   */

  if (
    partitionKeys.includes(previousValue)
  ) {

    queryPartition.value =
      previousValue;

  }

}

/* =========================================================
   SORT OPTIONS
========================================================= */

function updateSortOptions() {

  querySort.innerHTML = "";


  const selectedUserId =
    queryPartition.value;


  /*
   * Only show sort-key values that
   * actually belong to the selected
   * partition.
   */

  const matchingUsers =
    users.filter(user =>
      user.UserId === selectedUserId
    );


  /* -----------------------------------------
     Base table
  ------------------------------------------ */

  if (
    currentSource === "table"
  ) {

    sortLabel.textContent =
      "Sort Key";

    sortDescription.textContent =
      "CreatedAt";


    matchingUsers.forEach(user => {

      const option =
        document.createElement(
          "option"
        );

      option.value =
        user.CreatedAt;

      option.textContent =
        user.CreatedAt;

      querySort.appendChild(
        option
      );

    });

  }


  /* -----------------------------------------
     LSI
  ------------------------------------------ */

  if (
    currentSource === "lsi"
  ) {

    sortLabel.textContent =
      "LSI Sort Key";

    sortDescription.textContent =
      "Email";


    matchingUsers.forEach(user => {

      const option =
        document.createElement(
          "option"
        );

      option.value =
        user.Email;

      option.textContent =
        user.Email;

      querySort.appendChild(
        option
      );

    });

  }

}


/* =========================================================
   QUERY
========================================================= */

function runQuery() {

  clearHighlights();


  const userId =
    queryPartition.value;

  const sortValue =
    querySort.value;


  visualUserId.textContent =
    userId;

  visualSortValue.textContent =
    sortValue;


  let result;


  /* -----------------------------------------
     Base table query
  ------------------------------------------ */

  if (
    currentSource === "table"
  ) {

    visualSortKey.textContent =
      "CreatedAt";


    result =
      users.find(user =>

        user.UserId === userId &&
        user.CreatedAt === sortValue

      );


    queryCode.textContent =
`Query({
  TableName: "Users",
  KeyConditionExpression:
    "UserId = :userId AND CreatedAt = :createdAt",
  ExpressionAttributeValues: {
    ":userId": "${userId}",
    ":createdAt": "${sortValue}"
  }
})`;

  }


  /* -----------------------------------------
     LSI query
  ------------------------------------------ */

  if (
    currentSource === "lsi"
  ) {

    visualSortKey.textContent =
      "Email";


    result =
      users.find(user =>

        user.UserId === userId &&
        user.Email === sortValue

      );


    queryCode.textContent =
`Query({
  TableName: "Users",
  IndexName: "EmailIndex",
  KeyConditionExpression:
    "UserId = :userId AND Email = :email",
  ExpressionAttributeValues: {
    ":userId": "${userId}",
    ":email": "${sortValue}"
  }
})`;

  }


  /* -----------------------------------------
     Highlight result
  ------------------------------------------ */

  if (result) {

    usersRows.forEach(row => {

      const matches =
        row.dataset.id ===
          result.UserId &&
        (
          currentSource === "table"
            ? row.dataset.created ===
                result.CreatedAt
            : row.dataset.email ===
                result.Email
        );


      if (matches) {

        row.classList.add(
          "row-highlight"
        );

      }

    });


    /*
     * When querying the LSI,
     * highlight the matching index row too.
     */

    if (
      currentSource === "lsi"
    ) {

      indexRows.forEach(row => {

        if (
          row.dataset.id ===
            result.UserId &&
          row.dataset.email ===
            result.Email
        ) {

          row.classList.add(
            "row-highlight"
          );

        }

      });

    }


    queryResults.textContent =
      formatJSON(result);

  }
  else {

    queryResults.textContent =
      "No matching item.";

  }

}

/* =========================================================
   SCAN
========================================================= */

function runScan() {

  clearHighlights();


  const filter =
    scanFilter.value;


  let results;


  /* -----------------------------------------
     No filter
  ------------------------------------------ */

  if (!filter) {

    results =
      [...users];


    usersRows.forEach(row => {

      row.classList.add(
        "row-scanned"
      );

    });


    scanCode.textContent =
`Scan({
  TableName: "Users"
})`;

  }


  /* -----------------------------------------
     Filter
  ------------------------------------------ */

  else {

    results =
      users.filter(user =>
        user.Email === filter
      );


    usersRows.forEach(row => {

      row.classList.add(
        "row-scanned"
      );


      if (
        row.dataset.email ===
        filter
      ) {

        row.classList.remove(
          "row-scanned"
        );

        row.classList.add(
          "row-highlight"
        );

      }
      else {

        row.classList.add(
          "row-filtered"
        );

      }

    });


    scanCode.textContent =
`Scan({
  TableName: "Users",
  FilterExpression:
    "Email = :email",
  ExpressionAttributeValues: {
    ":email": "${filter}"
  }
})`;

  }


  scanResultCount.textContent =
    `${results.length} result${
      results.length === 1
        ? ""
        : "s"
    }`;


  scanResults.textContent =
    formatJSON(results);

}
