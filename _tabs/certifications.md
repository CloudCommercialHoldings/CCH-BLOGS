<script>
  (function () {
    // 1. Calculate dynamic angles for all badges
    const badges = document.querySelectorAll('.cert-badge-item');
    const total = badges.length;
    badges.forEach((badge, index) => {
      const angle = (index * 360) / total;
      badge.style.setProperty('--angle', `${angle}deg`);
    });

    // 2. Load serialized certifications JSON data
    const certsData = {{ page.certs | jsonify }};
    const modal = document.getElementById("certModal");
    const modalBody = document.getElementById("modalBody");
    const closeBtn = document.getElementById("modalCloseBtn");

    // 3. Click handler to open details card
    badges.forEach(badge => {
      badge.addEventListener("click", () => {
        const id = parseInt(badge.getAttribute("data-id"), 10);
        const cert = certsData.find(c => c.id === id);
        if (cert) {
          openCertDetails(cert);
        }
      });
    });

    function openCertDetails(cert) {
      const focusHtml = cert.focus.map(f => `<li><i class="fas fa-check text-success me-2"></i>${f}</li>`).join('');
      const vulnsHtml = cert.vulns.map(v => `<li><i class="fas fa-exclamation-triangle text-warning me-2"></i>${v}</li>`).join('');
      const toolsHtml = cert.tools.map(t => `<span class="badge bg-secondary-subtle text-light border border-secondary px-2.5 py-1.5 rounded-pill small m-1">${t}</span>`).join('');

      modalBody.innerHTML = `
        <div class="d-flex align-items-center gap-3 mb-4 text-start">
          <svg width="40" height="40" viewBox="0 0 40 40" class="flex-shrink-0">
            <rect width="40" height="40" rx="8" fill="${cert.color || '#0078D4'}"></rect>
            <path d="M12 20l5 5 11-11" stroke="#fff" stroke-width="3" fill="none" stroke-linecap="round" stroke-linejoin="round"></path>
          </svg>
          <div>
            <h2 class="h5 fw-bold text-white mb-0">${cert.code}</h2>
            <p class="text-sky mb-0 small fw-semibold">${cert.title}</p>
          </div>
        </div>

        <div class="row g-4 my-2 text-start">
          <div class="col-md-6">
            <h3 class="h6 fw-bold text-sky mb-2"><i class="fas fa-shield-alt me-2"></i>Blue Team Focus</h3>
            <ul class="list-unstyled text-light-slate small mb-0">
              ${focusHtml}
            </ul>
          </div>
          
          <div class="col-md-6">
            <h3 class="h6 fw-bold text-danger mb-2"><i class="fas fa-bug me-2"></i>Vulnerability Analysis</h3>
            <ul class="list-unstyled text-light-slate small mb-0">
              ${vulnsHtml}
            </ul>
          </div>
        </div>

        <hr class="border-secondary my-4">

        <div class="mb-4 text-start">
          <h3 class="h6 fw-bold text-success mb-3"><i class="fas fa-tools me-2"></i>Security Tools</h3>
          <div class="d-flex flex-wrap gap-1">
            ${toolsHtml}
          </div>
        </div>

        <div class="d-flex justify-content-end gap-2 mt-4">
          <button class="btn btn-outline-light btn-sm px-4 rounded-2" id="modalCloseActionBtn">Close</button>
        </div>
      `;

      modal.classList.remove("hidden");
      document.getElementById("modalCloseActionBtn").addEventListener("click", closeCertDetails);
    }

    function closeCertDetails() {
      modal.classList.add("hidden");
    }

    // Modal close events
    closeBtn.addEventListener("click", closeCertDetails);
    modal.addEventListener("click", (e) => {
      if (e.target === modal) {
        closeCertDetails();
      }
    });

    document.addEventListener("keydown", (e) => {
      if (e.key === "Escape" && !modal.classList.contains("hidden")) {
        closeCertDetails();
      }
    });
  })();
</script>
